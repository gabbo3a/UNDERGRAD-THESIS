% Preamble: UniTO presentation style configuration (LaTeX Beamer)

% Encoding and language
\usepackage[utf8]{inputenc}
\usepackage[T1]{fontenc}
\usepackage[italian]{babel}

% Auxiliary packages
\usepackage{amsmath, amssymb}
\usepackage{graphicx}
\usepackage{booktabs}
\usepackage{xcolor}
\usepackage{listings}
\usepackage{bookmark}

% Presentation typography (Helvetica / Arial clone)
\usepackage[scaled=0.92]{helvet}
    \renewcommand{\familydefault}{\sfdefault}

% Graphics and local asset search path
\usepackage{graphicx}
\graphicspath{{assets/}{../assets/}}

% UniTO institutional colors
\definecolor{unitored}{RGB}{218, 24, 56}         % UniTO primary brand red
\definecolor{unitoslate}{RGB}{30, 41, 59}        % Primary slate text
\definecolor{unitogray}{RGB}{246, 247, 249}      % Neutral block/code background
\definecolor{thanksgray}{HTML}{444B52}            % Charcoal gray for subtitles and "Grazie"

% Global beamer settings
\usetheme{default}
\setbeamertemplate{navigation symbols}{}

% Background and color hierarchy
\setbeamercolor{background canvas}{bg=white}
\setbeamercolor{normal text}{fg=unitoslate}
\setbeamercolor{structure}{fg=unitored}

% Subtitle styling (subtle gray)
\setbeamercolor{subtitle}{fg=thanksgray}
\setbeamerfont{subtitle}{size=\normalsize, series=\normalfont}

% Slide header (slate color matching main title, smaller size, larger logo on the right, no lines)
\setbeamercolor{frametitle}{fg=unitoslate, bg=white}
\setbeamertemplate{frametitle}{%
    \vskip 0.20cm%
    \begin{columns}[c, onlytextwidth]%
        \begin{column}{0.68\textwidth}%
            {\fontsize{13pt}{15pt}\selectfont\bfseries\color{unitoslate}\insertframetitle}%
        \end{column}%
        \begin{column}{0.31\textwidth}%
            \raggedleft\includegraphics[height=1.45cm]{unito.png}%
        \end{column}%
    \end{columns}%
    \vskip 0.15cm%
}

% Footer (page number only, author removed)
\setbeamertemplate{footline}{%
    \leavevmode%
    \hbox{%
        \begin{beamercolorbox}[wd=\paperwidth,ht=2.5ex,dp=1ex,right]{date in head/foot}%
            \scriptsize\color{gray}\insertframenumber{} / \inserttotalframenumber\hspace*{3ex}%
        \end{beamercolorbox}%
    }%
    \vskip 0.15cm%
}

% Block environments
\setbeamercolor{block title}{fg=white, bg=unitored}
\setbeamercolor{block body}{fg=unitoslate, bg=unitogray}

% Lean 4 syntax highlighting
\lstdefinelanguage{lean4}{
    keywords={inductive, def, theorem, where, match, with, by, sorry, have, rw, exact, intro},
    keywordstyle=\color{unitored}\bfseries,
    sensitive=true,
    comment=[l]{--},
    commentstyle=\color{gray}\itshape,
    basicstyle=\ttfamily\scriptsize,
    backgroundcolor=\color{unitogray},
    frame=single,
    rulecolor=\color{gray!30}
}
